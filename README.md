async function compressImage(file){
  const dataUrl=await new Promise((resolve,reject)=>{const reader=new FileReader();reader.onload=()=>resolve(reader.result);reader.onerror=reject;reader.readAsDataURL(file)});
  const img=await new Promise((resolve,reject)=>{const image=new Image();image.onload=()=>resolve(image);image.onerror=reject;image.src=dataUrl});
  const canvas=document.createElement('canvas');const scale=Math.min(1,900/Math.max(img.width,img.height));canvas.width=Math.round(img.width*scale);canvas.height=Math.round(img.height*scale);
  canvas.getContext('2d').drawImage(img,0,0,canvas.width,canvas.height);
  let quality=.72,result=canvas.toDataURL('image/jpeg',quality);
  while(result.length>450000&&quality>.35){quality-=.08;result=canvas.toDataURL('image/jpeg',quality)}
  return result
}

## Railway environment variables
Required: `BOT_TOKEN`, `ADMIN_PASSWORD`, `WEBHOOK_SECRET`, `RAILWAY_PUBLIC_DOMAIN`, `ADMIN_TELEGRAM_ID`, `PLATFORM_API_URL`, `BOT_TRIAL_SECRET`.
Optional: `ADMIN_USER` (defaults to admin), `OPENAI_API_KEY` for image matching/receipt visual analysis, `DATA_FILE` (defaults to `/data/replies.json`).

Set the same `BOT_TRIAL_SECRET` on both the bot service and platform service. `PLATFORM_API_URL` must be the platform server base URL, not the Firebase RTDB URL. Make sure the Telegram account is connected to the bot as a Business bot and that webhook updates include `business_message` and `business_connection`.

## Deploy
Replace the bot service `server.js`, `public/index.html`, and `package.json` with these files, then redeploy. Keep your existing persistent volume mounted at `/data` so saved replies, image references and payment requests persist.

## Notes
- Payment receipt image similarity is not proof that money arrived. Admin must verify the actual transfer before approving.
- Image-reference matching and visual receipt analysis require `OPENAI_API_KEY`; if not configured, normal payment flow and keyword replies still work, but visual matching is disabled.
- The bot cannot read old Telegram chat history. One-time trial prevention is enforced by the platform's `botStudentLinks` record for Telegram user ID.
- Trial code generation requires the platform route `/api/bot/trial`; payment approval requires `/api/bot/payment/approve`.
